' À COLLER DANS LE MODULE DE LA FEUILLE (pas dans un module standard) :
' clic droit sur l'onglet de la feuille > Visualiser le code, puis coller.
' - Niveau d'étage (C1) modifié : les étages affichés sont mis à jour.
' - Type de mur modifié en B : la liste des épaisseurs en F est mise à jour.
' - Épaisseur choisie en F (ex. "20 cm") : convertie en mètres (0.20).

Private Sub Worksheet_Change(ByVal Target As Range)
    Dim c As Range

    If Not Intersect(Target, Me.Range("C1")) Is Nothing Then AfficherEtages Me
    If Intersect(Target, Me.Range("B5:B32,F5:F32")) Is Nothing Then Exit Sub

    Application.EnableEvents = False
    For Each c In Intersect(Target, Me.Range("B5:B32,F5:F32")).Cells
        If EpaisseursMur(CStr(Me.Range("B" & c.Row).Value)) <> "" Then
            If c.Column = 2 Then
                MajEpaisseurMur Me, c.Row
            ElseIf VarType(c.Value) = vbString Then
                c.Value = Val(c.Value) / 100
            End If
        End If
    Next c
    Application.EnableEvents = True
End Sub
